/**
 * Mencatat riwayat aktivitas ke dalam Sheet "Log_Aktivitas"
 */
function writeLog(action, details, username, ss) {
  try {
    const spreadsheet = ss || SpreadsheetApp.getActiveSpreadsheet();
    let sheet = spreadsheet.getSheetByName("Log_Aktivitas");
    if (!sheet) {
      // Buat sheet baru secara otomatis jika belum ada di database
      sheet = ss.insertSheet("Log_Aktivitas");
      sheet.appendRow(["WAKTU", "PENGGUNA", "AKSI", "DETAIL"]);
      sheet.getRange("A1:D1").setFontWeight("bold").setBackground("#fca5a5").setHorizontalAlignment("center");
    }
    const timestamp = Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd HH:mm:ss");
    sheet.appendRow([timestamp, username || "Guest/System", action, details]);
  } catch (e) {
    Logger.log("Gagal merekam log aktivitas: " + e.message);
  }
}

/**
 * Mengambil 200 riwayat log aktivitas terbaru dari sheet Log_Aktivitas
 */
function getLogs() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Log_Aktivitas");
    if (!sheet) return [];
    
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return [];
    
    const headers = data[0];
    const logs = data.slice(1).map(function(row) {
      let obj = {};
      headers.forEach(function(h, i) {
        let key = String(h).trim().toUpperCase();
        let val = row[i];
        if (val instanceof Date) {
          val = Utilities.formatDate(val, "GMT+7", "yyyy-MM-dd HH:mm:ss");
        }
        obj[key] = val;
      });
      return obj;
    });
    
    // Sort log dari waktu terbaru ke terlama
    return logs.reverse().slice(0, 200);
  } catch (e) {
    throw new Error("Gagal mengambil log sistem: " + e.message);
  }
}

function loginUser(username, password) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Users");
    
    if (!sheet) {
      return { success: false, message: "Kesalahan Sistem: Sheet 'Users' tidak ditemukan. Pastikan nama sheet benar." };
    }
    
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(username).trim() && String(data[i][1]).trim() === String(password).trim()) {
        // Tulis log sukses login
        writeLog("LOGIN", "Berhasil masuk ke dalam aplikasi PAC Kawunganten", username);
        return { success: true, role: data[i][2], username: username };
      }
    }
    writeLog("LOGIN_FAILED", "Mencoba masuk dengan password yang keliru", username);
    return { success: false, message: "Username atau Password salah atau tidak terdaftar!" };
  } catch (e) {
    return { success: false, message: "Error Internal Server: " + e.message };
  }
}

const _folderCache = {};
function getFolderByName(parentFolder, folderName) {
  var cacheKey = parentFolder.getId() + "_" + folderName;
  if (_folderCache[cacheKey]) return _folderCache[cacheKey];

  var props = PropertiesService.getScriptProperties();
  var cachedId = props.getProperty(cacheKey);
  
  if (cachedId) {
    try {
      var folder = DriveApp.getFolderById(cachedId);
      _folderCache[cacheKey] = folder;
      return folder;
    } catch(e) {
      props.deleteProperty(cacheKey);
    }
  }

  var folders = parentFolder.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : parentFolder.createFolder(folderName);
  
  props.setProperty(cacheKey, folder.getId());
  _folderCache[cacheKey] = folder;
  return folder;
}

function getAppData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetAnggota = ss.getSheetByName("Data_Anggota");
    const dataAnggota = sheetAnggota.getDataRange().getValues();
    let anggotaList = [];
    let stats = { total: 0, pac: 0, ranting: 0, anakRanting: 0, satgas: 0 };
    
    if (dataAnggota.length > 1) {
      const headers = dataAnggota[0];
      anggotaList = dataAnggota.slice(1).map(function(row) {
        let obj = {};
        headers.forEach(function(h, i) {
          let key = String(h).trim().replace(/\s+/g, '_').toUpperCase();
          let val = row[i];
          if (val instanceof Date) {
            let day = ("0" + val.getDate()).slice(-2);
            let month = ("0" + (val.getMonth() + 1)).slice(-2);
            let year = val.getFullYear();
            val = year + '-' + month + '-' + day;
          } else {
            val = String(val).trim();
          }
          obj[key] = val;
        });
        return obj;
      });
      anggotaList.forEach(function(item) {
        stats.total++;
        if (item.BAGIAN === "PAC") stats.pac++;
        else if (item.BAGIAN === "RANTING") stats.ranting++;
        else if (item.BAGIAN === "ANAK RANTING") stats.anakRanting++;
        else if (item.BAGIAN === "SATGAS") stats.satgas++;
      });
    }

    const sheetWilayah = ss.getSheetByName("Wilayah");
    let wilayahData = [];
    if (sheetWilayah) {
      const dataWilayah = sheetWilayah.getDataRange().getValues();
      if (dataWilayah.length > 1) {
        wilayahData = dataWilayah.slice(1).map(function(r) { return { kecamatan: String(r[0]).trim(), desa: String(r[1]).trim(), dusun: String(r[2]).trim() }; });
      }
    }
    return { anggota: anggotaList, stats: stats, wilayah: wilayahData };
  } catch (e) {
    throw new Error("Gagal memuat data: " + e.message);
  }
}

function saveAnggota(formData, fotoKtpBase64, passFotoBase64, operator) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Data_Anggota");
  const rootFolder = DriveApp.getFolderById(FOLDER_ID);
  
  const lastRow = Math.max(sheet.getLastRow(), 1);
  const nextId = lastRow === 1 ? 1001 : Number(sheet.getRange(lastRow, 1).getValue()) + 1;
  const namaDepan = formData.nama.split(' ')[0];
  const fileName = nextId + '_' + namaDepan;

  const saveFile = function(base64, tipeFile) {
    if (!base64) return "";
    
    const namaDesa = formData.desa ? String(formData.desa).trim() : "TANPA_DESA";
    const namaBagian = formData.bagian ? String(formData.bagian).trim() : "TANPA_BAGIAN";
    
    const folderTipe = getFolderByName(rootFolder, tipeFile);
    const folderDesa = getFolderByName(folderTipe, namaDesa);
    const folderBagian = getFolderByName(folderDesa, namaBagian);
    
    const blob = Utilities.newBlob(Utilities.base64Decode(base64.split(',')[1]), "image/jpeg", fileName + ".jpg");
    const file = folderBagian.createFile(blob);
    return file.getUrl();
  };

  const fotoKtpUrl = saveFile(fotoKtpBase64, "FOTO_KTP");
  const passFotoUrl = saveFile(passFotoBase64, "PASS_FOTO");

  let age = 0;
  if (formData.tanggalLahir) {
      const birthDate = new Date(formData.tanggalLahir);
      age = new Date().getFullYear() - birthDate.getFullYear();
  }

  sheet.appendRow([nextId, "'" + formData.nik, formData.nama, formData.tanggalLahir, formData.jenisKelamin, age, "'" + formData.nomorHp, formData.bagian, formData.jabatan, formData.kecamatan, formData.desa, formData.dusun, fotoKtpUrl, passFotoUrl]);
  
  // Catat ke log aktivitas
  writeLog("TAMBAH_ANGGOTA", "Menambahkan anggota baru: " + formData.nama + " (ID: " + nextId + ")", operator, ss);
  return { success: true, id: nextId };
}

function updateAnggota(id, formData, fotoKtpBase64, passFotoBase64, operator) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Data_Anggota");
  const data = sheet.getDataRange().getValues();
  const rootFolder = DriveApp.getFolderById(FOLDER_ID);
  const namaDepan = formData.nama.split(' ')[0];
  const fileName = id + '_' + namaDepan + '_update';
  
  const updateFile = function(base64, tipeFile, oldUrl) {
    if (!base64) return oldUrl;
    
    const namaDesa = formData.desa ? String(formData.desa).trim() : "TANPA_DESA";
    const namaBagian = formData.bagian ? String(formData.bagian).trim() : "TANPA_BAGIAN";
    
    const folderTipe = getFolderByName(rootFolder, tipeFile);
    const folderDesa = getFolderByName(folderTipe, namaDesa);
    const folderBagian = getFolderByName(folderDesa, namaBagian);
    
    const blob = Utilities.newBlob(Utilities.base64Decode(base64.split(',')[1]), "image/jpeg", fileName + ".jpg");
    const file = folderBagian.createFile(blob);
    return file.getUrl();
  };

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(id).trim()) {
      let fotoKtpUrl = updateFile(fotoKtpBase64, "FOTO_KTP", data[i][12]);
      let passFotoUrl = updateFile(passFotoBase64, "PASS_FOTO", data[i][13]);
      
      let age = 0;
      if(formData.tanggalLahir){
          const birthDate = new Date(formData.tanggalLahir);
          age = new Date().getFullYear() - birthDate.getFullYear();
      }
      
      sheet.getRange(i + 1, 2, 1, 13).setValues([[
        "'" + formData.nik, formData.nama, formData.tanggalLahir, formData.jenisKelamin, 
        age, "'" + formData.nomorHp, formData.bagian, formData.jabatan, formData.kecamatan, 
        formData.desa, formData.dusun, fotoKtpUrl, passFotoUrl
      ]]);
      
      // Catat ke log aktivitas
      writeLog("UPDATE_ANGGOTA", "Mengubah profil data anggota: " + formData.nama + " (ID: " + id + ")", operator, ss);
      return { success: true };
    }
  }
  return { success: false };
}

/**
 * ========================================================
 * MENU 2: FUNGSI MANAJEMEN KAS ORGANISASI
 * ========================================================
 */
function getKasData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Kas_Organisasi");
    if (!sheet) {
      sheet = ss.insertSheet("Kas_Organisasi");
      sheet.appendRow(["ID", "TANGGAL", "TIPE", "NOMINAL", "KATEGORI", "KETERANGAN", "OPERATOR"]);
      sheet.getRange("A1:G1").setFontWeight("bold").setBackground("#dc2626").setColor("#ffffff").setHorizontalAlignment("center");
    }
    
    const data = sheet.getDataRange().getValues();
    let listKas = [];
    let summary = { masuk: 0, keluar: 0, saldo: 0 };
    
    if (data.length > 1) {
      const headers = data[0];
      listKas = data.slice(1).map(function(row) {
        let obj = {};
        headers.forEach(function(h, i) {
          let key = String(h).trim().toUpperCase();
          let val = row[i];
          if (val instanceof Date) {
            let day = ("0" + val.getDate()).slice(-2);
            let month = ("0" + (val.getMonth() + 1)).slice(-2);
            let year = val.getFullYear();
            val = year + '-' + month + '-' + day; // Format YYYY-MM-DD
          }
          obj[key] = val;
        });
        
        // Konversi nominal
        const nominalValue = Number(obj.NOMINAL) || 0;
        if (String(obj.TIPE).toUpperCase() === "PEMASUKAN") {
          summary.masuk += nominalValue;
        } else if (String(obj.TIPE).toUpperCase() === "PENGELUARAN") {
          summary.keluar += nominalValue;
        }
        
        return obj;
      });
    }
    
    summary.saldo = summary.masuk - summary.keluar;
    
    // Sort transaksi dari tanggal terbaru ke terlama
    listKas.sort(function(a, b) {
      return new Date(b.TANGGAL) - new Date(a.TANGGAL);
    });
    
    return { kas: listKas, summary: summary };
  } catch (e) {
    throw new Error("Gagal memuat kas organisasi: " + e.message);
  }
}
